Code 
colnames(arima)[2:13] <- paste0(colnames(arima)[2:13], "_arima")
colnames(acp)[2:13] <- paste0(colnames(acp)[2:13], "_acp")
merged_data <- merge(arima, acp, by = "date")
ordered_columns <- c("date")
for (i in 1:12) {
  ordered_columns <- c(ordered_columns, paste0("B", i, "_arima"), paste0("B", i, "_acp"))
}
merged_data_ordered <- merged_data[, ordered_columns]
for (i in 1:12) {
  B_arima <- paste0("B", i, "_arima")
  B_acp <- paste0("B", i, "_acp")
    df_long <- data.frame(
    date = rep(as.Date(paste0(merged_data_ordered$date, "-01"), format = "%Y_%m-%d"), 2),
    value = c(merged_data_ordered[[B_arima]], merged_data_ordered[[B_acp]]),
    method = rep(c("ARIMA", "ACP"), each = nrow(merged_data_ordered))
  )
  p <- ggplot(df_long, aes(x = date, y = value, color = method, group = method)) +
    geom_line() +
    labs(
      title = paste("Evolution des prix -B", i),
      x = "Date", 
      y = "Valeur", 
      color = "Method"
    ) +
    theme_bw() +
    theme(axis.text.x = element_text(angle = 90, hjust = 1)) +
    scale_x_date(date_labels = "%b %Y", date_breaks = "2 months", expand = c(0, 0)) +
    scale_color_manual(values = c("blue", "red"))
  print(p)
}
